function doGet(e) { try { var action = (e && e.parameter && e.parameter.action) || ""; if (action === "list") { return jsonOut_({ status: "success", data: listApplications_(e.parameter.userId || "") }); } if (action === "listAll" || action === "getAll" || action === "all") { // Kembalikan SEMUA permohonan dari SEMUA sheet/kategori return jsonOut_({ status: "success", data: listAllApplications_() }); } return jsonOut_({ status: "error", message: "Unknown action: " + action }); } catch (err) { return jsonOut_({ status: "error", message: err.toString() }); } } /** * Senarai SEMUA permohonan dari semua sheet. * Sesuaikan senarai nama sheet di bawah mengikut spreadsheet anda. */ function listAllApplications_() { var ss = SpreadsheetApp.openById("SPREADSHEET_ID_ANDA"); var sheets = ss.getSheets(); var all = []; for (var s = 0; s < sheets.length; s++) { var sheet = sheets[s]; var name = sheet.getName(); // Skip sheet yang bukan permohonan if (name === "Config" || name === "Log" || name === "Users") continue; var rows = readSheet_(sheet); for (var i = 0; i < rows.length; i++) { var r = rows[i]; // Tag dengan nama sheet sebagai kategori r.category = name; r.sheet = name; all.push(r); } } // Susun mengikut tarikh hantar (terbaru dulu) all.sort(function(a, b) { return new Date(b.submittedAt || 0) - new Date(a.submittedAt || 0); }); return all; } function readSheet_(sheet) { var values = sheet.getDataRange().getValues(); if (values.length < 2) return []; var headers = values[0]; var out = []; for (var i = 1; i < values.length; i++) { var row = {}; for (var j = 0; j < headers.length; j++) { row[headers[j]] = values[i][j]; } // Parse JSON fields jika ada ["items", "shared"].forEach(function(field) { if (typeof row[field] === "string" && row[field]) { try { row[field] = JSON.parse(row[field]); } catch(e) {} } }); if (row.ref) out.push(row); } return out; } function jsonOut_(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
